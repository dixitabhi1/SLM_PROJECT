from sqlalchemy import create_engine, Column, Integer, String, Date, Text, ForeignKey
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker, relationship

Base = declarative_base()

class Turbine(Base):
    __tablename__ = 'turbines'
    id = Column(Integer, primary_key=True)
    name = Column(String, nullable=False)
    model = Column(String, nullable=False)
    manufacturer = Column(String, nullable=False)
    components = relationship("Component", back_populates="turbine")

class Component(Base):
    __tablename__ = 'components'
    id = Column(Integer, primary_key=True)
    turbine_id = Column(Integer, ForeignKey('turbines.id'))
    name = Column(String, nullable=False)
    description = Column(String)
    part_number = Column(String, unique=True)
    turbine = relationship("Turbine", back_populates="components")

class InspectionLog(Base):
    __tablename__ = 'inspection_logs'
    id = Column(Integer, primary_key=True)
    component_id = Column(Integer, ForeignKey('components.id'))
    inspection_date = Column(Date, nullable=False)
    inspection_type = Column(String, nullable=False)
    results = Column(String)
    notes = Column(String)
    component = relationship("Component", back_populates="logs")
